1"""Regenerate the committed shared baseline artefact (tests/eval/baseline.json) from one build. 

2 

3Builds the shared fixture once via `_build_populated_fixture`, measures both the ranking 

4metrics and the read-tool output sizes against it, and writes both sections of the artefact 

5in a single pass via each domain module's own `section()` builder. This is the sole entry 

6point for baseline regeneration - `just baseline` (and its `--rebaseline` flag) always go 

7through this module. 

8 

9Run with no arguments to CHECK the committed artefact against the current code; run with 

10`--rebaseline` to overwrite it. Overwriting is deliberately a separate act: the gate is only 

11meaningful while the artefact predates the run being judged. 

12""" 

13 

14from __future__ import annotations 

15 

16import argparse 

17import json 

18from pathlib import Path 

19import tempfile 

20from typing import TYPE_CHECKING 

21 

22from mcp_memory.storage import open_writable 

23from tests.eval.eval_baseline import ( 

24 BASELINE_PATH, 

25 Baseline, 

26 format_moves as format_ranking_moves, 

27 load_baseline as load_ranking, 

28 moved as ranking_moved, 

29 section as section_ranking, 

30) 

31from tests.eval.eval_fixture import _build_populated_fixture 

32from tests.eval.size_baseline import ( 

33 SizeBaseline, 

34 format_moves as format_size_moves, 

35 load_baseline as load_size, 

36 measure as measure_size, 

37 moved as size_moved, 

38 section as section_size, 

39) 

40 

41if TYPE_CHECKING: 

42 from collections.abc import Callable, Sequence 

43 

44 

45def _measure_both() -> tuple[Baseline, SizeBaseline]: 

46 """Build the shared fixture once and derive both the ranking and size baselines from it.""" 

47 with tempfile.TemporaryDirectory() as tmp: 

48 db = open_writable(Path(tmp) / "baseline.db") 

49 try: 

50 fixture = _build_populated_fixture(db) 

51 return Baseline.from_report(fixture.expected_baseline), measure_size(fixture) 

52 finally: 

53 db.connection.close() 

54 

55 

56def _write_combined(path: Path, ranking: Baseline, size: SizeBaseline) -> None: 

57 """Write both sections to `path` in one pass, each section built by its own module.""" 

58 combined = {"ranking": section_ranking(ranking), "size": section_size(size)} 

59 path.write_text(json.dumps(combined, indent=2) + "\n", encoding="utf-8") 

60 

61 

62def build_parser() -> argparse.ArgumentParser: 

63 """Build the argument parser for the regeneration command.""" 

64 parser = argparse.ArgumentParser(description=__doc__) 

65 parser.add_argument( 

66 "--rebaseline", 

67 action="store_true", 

68 help="overwrite the committed baseline with the measured values", 

69 ) 

70 return parser 

71 

72 

73def main( 

74 argv: Sequence[str] | None = None, 

75 *, 

76 path: Path, 

77 measure: Callable[[], tuple[Baseline, SizeBaseline]] = _measure_both, 

78) -> int: 

79 """Check both sections of the baseline at `path` against a fresh measurement, or rewrite it. 

80 

81 `path` is keyword-only with no default so a caller cannot reach the committed artefact by 

82 omission - a test that rewrote it with synthetic values would silently make the gate 

83 decorative. Both sections are always measured and, if written, written together, so 

84 neither can be regenerated without the other. 

85 """ 

86 args = build_parser().parse_args(argv) 

87 bootstrap = args.rebaseline and not path.exists() 

88 committed_ranking = None if bootstrap else load_ranking(path) 

89 committed_size = None if bootstrap else load_size(path) 

90 measured_ranking, measured_size = measure() 

91 if committed_ranking is None or committed_size is None: 

92 _write_combined(path, measured_ranking, measured_size) 

93 print(f"wrote {path} (first baseline)") 

94 return 0 

95 ranking_drift = ranking_moved(committed_ranking, measured_ranking) 

96 size_drift = size_moved(committed_size, measured_size) 

97 if ranking_drift: 

98 print(format_ranking_moves(committed_ranking, measured_ranking)) 

99 else: 

100 print("ranking baseline unchanged") 

101 if size_drift: 

102 print(format_size_moves(committed_size, measured_size)) 

103 else: 

104 print("size baseline unchanged") 

105 if not ranking_drift and not size_drift: 

106 return 0 

107 if not args.rebaseline: 

108 sections = [("ranking", ranking_drift), ("size", size_drift)] 

109 drifted = [name for name, drift in sections if drift] 

110 print(f"baseline drift in {drifted}; run `just baseline --rebaseline` to update it") 

111 return 1 

112 lowered = sorted(metric for metric, (old, new) in ranking_drift.items() if new < old) 

113 if lowered: 

114 print(f"LOWERED: {lowered} - a lowered bound weakens the gate; justify it in the commit") 

115 increased = sorted(tool for tool, (old, new) in size_drift.items() if new > old) 

116 if increased: 

117 print(f"INCREASED: {increased} - a larger output costs more context on every call; justify it in the commit") 

118 _write_combined(path, measured_ranking, measured_size) 

119 print(f"wrote {path}") 

120 return 0 

121 

122 

123if __name__ == "__main__": 

124 raise SystemExit(main(path=BASELINE_PATH))
