1"""Anonymous usage telemetry, sent through the helpers llm-prompts shares. 

2 

3Off when llm-prompts is not installed or its opt-out applies. Only tool names, counts, byte sizes, ranking-score 

4means, power-of-ten bucketed counts and version identifiers (commits of the installed 

5llm-prompts, cline-hooks and mcp-memory) leave the machine: never the stored options, queries, entity or 

6project names or memory text. 

7 

8The ``memory_usage``, ``memory_eval``, ``memory_dream`` and crash events carry ``versions`` (for llm-prompts, 

9cline-hooks and mcp-memory, each {commit, base, local, dirty, ahead}, where ``base`` is the newest origin/main commit 

10the install contains) and ``overlays`` (the same for overlay installs from GitHub). 

11 

12The ``memory_usage`` event also carries the counts ``searches``, ``searches_used`` (searches with a result that was 

13used afterwards), ``empty_searches``, ``writes`` and ``nudges`` (reminders the hooks showed, also by kind in 

14``nudges_by_kind``), so real use can be compared across versions. ``relations_created`` counts the links made, by 

15type (any valid relation type, else ``other``), and the ``relations`` and 

16``linked_entities`` size buckets show how connected the store is. 

17 

18The ``memory_eval`` event is sent each time the server goes idle after new searches. It carries the score means, the 

19``db_mb``, ``entities`` and ``projects`` size buckets and ``min_content_tokens``, the shortest query, in words, that 

20was scored. 

21"""

125def _exception_event(core: ModuleType, exc: BaseException, where: str) -> dict[str, Any] | None: 

126 """Build a crash event unless this failure site was already reported.""" 

127 event: dict[str, Any] = core.exception_event(exc, {}, {"where": where, **_context(core)}) 

128 items = event["properties"]["$exception_list"] 

129 frames = items[0]["stacktrace"]["frames"] 

130 site = (items[0]["type"], *((frames[-1]["function"], frames[-1]["lineno"]) if frames else ())) 

131 if site in _reported_failures: 

132 return None 

133 _reported_failures.add(site) 

134 return event

202 def _rollup(self, up_to: int | None = None) -> tuple[dict[str, dict[str, int | float]], int]: 

203 """Return per-tool counts and median sizes since the last report, and the newest call id they cover. 

204 

205 With ``up_to``, only calls up to that id count. 

206 """ 

207 rows = self._db.connection.query_all( 

208 "SELECT id, tool, input_bytes, output_bytes FROM tool_calls WHERE id > ? AND id <= ?", 

209 (self._last_id, self._newest_id() if up_to is None else up_to), 

210 ) 

211 calls: dict[str, list[Any]] = {} 

212 for row in rows: 

213 calls.setdefault(row["tool"], []).append(row) 

214 errors = {tool: count for tool, count in _errors.items() if count > 0} 

215 tools = { 

216 tool: { 

217 "ok": len(calls.get(tool, [])), 

218 "errors": errors.get(tool, 0), 

219 "median_input_bytes": statistics.median([r["input_bytes"] for r in calls.get(tool, [])] or [0]), 

220 "median_output_bytes": statistics.median([r["output_bytes"] for r in calls.get(tool, [])] or [0]), 

221 } 

222 for tool in sorted({*calls, *errors}) 

223 } 

224 return tools, max((row["id"] for row in rows), default=self._last_id) 

225 

226 def _surfaced_since_last_report(self, up_to: int | None) -> tuple[int, int, int]: 

227 """Return the searches with a used result, the searches with any result, and the newest result id. 

228 

229 Counts cover the results surfaced since the last report. With ``up_to``, only results surfaced by the time 

230 of that call count. 

231 """ 

232 rows = self._db.connection.query_all( 

233 f"SELECT id, retrieval_id, used_at IS NOT NULL AS used FROM surfaced_entities WHERE id > ? " 

234 f"AND surfaced_at <= {_UP_TO_CALL}", 

235 (self._last_surfaced_id, up_to), 

236 ) 

237 used = len({row["retrieval_id"] for row in rows if row["used"]}) 

238 answered = len({row["retrieval_id"] for row in rows}) 

239 return used, answered, max((row["id"] for row in rows), default=self._last_surfaced_id) 

240 

241 def _links_since_last_report(self, up_to: int | None) -> tuple[dict[str, int], int]: 

242 """Return the links created since the last report by type, and the newest link id. 

243 

244 Types outside the valid relation types count as ``other``. With ``up_to``, only links created by the time 

245 of that call count. 

246 """ 

247 rows = self._db.connection.query_all( 

248 f"SELECT r.id, t.name FROM relations r JOIN relation_types t ON t.id = r.relation_type_id " 

249 f"WHERE r.id > ? AND r.created_at <= {_UP_TO_CALL}", 

250 (self._last_relation_id, up_to), 

251 ) 

252 counts = Counter(row["name"] if row["name"] in VALID_RELATION_TYPES else _OTHER_LINK_TYPE for row in rows) 

253 return dict(counts), max((row["id"] for row in rows), default=self._last_relation_id)

255 def _deliver( 

256 self, 

257 core: ModuleType, 

258 tools: dict[str, dict[str, int | float]], 

259 counts: _WindowCounts, 

260 install_id: str, 

261 ) -> bool: 

262 """Post the rollup inline and return whether it arrived.""" 

263 searches = sum(tools[tool]["ok"] for tool in SEARCH_TOOLS if tool in tools) 

264 properties = { 

265 "tools": tools, 

266 "searches": searches, 

267 "searches_used": counts.searches_used, 

268 "empty_searches": max(searches - counts.searches_with_results, 0), 

269 "relations_created": counts.relations_created, 

270 "nudges": sum(counts.nudges_by_kind.values()), 

271 "nudges_by_kind": counts.nudges_by_kind, 

272 "writes": sum(tools[tool]["ok"] for tool in MUTATING_TOOLS if tool in tools), 

273 **_store_size(self._db, core), 

274 **_context(core), 

275 } 

276 return bool(core.post_event({"event": "memory_usage", "properties": properties}, install_id))

278 def note_call(self, idle_seconds: float) -> None:

280 if idle_seconds >= IDLE_SECONDS: 

281 self._new_use_cutoff = self._newest_id()

283 def report(self, up_to: int | None = None) -> None: 

284 """Send the calls since the last report, unless telemetry is off or there were none. 

285 

286 A failed send leaves the calls and error counts in place for the next report. 

287 With ``up_to``, calls after that id wait. Error counts are not tied to call ids, so all current ones are sent. 

288 """ 

289 with self._report_lock: 

290 tools, newest_id = self._rollup(up_to) 

291 searches_used, searches_with_results, newest_surfaced_id = self._surfaced_since_last_report(up_to) 

292 relations_created, newest_relation_id = self._links_since_last_report(up_to) 

293 counts = _WindowCounts( 

294 searches_used, searches_with_results, relations_created, nudge_count.read() if tools else {} 

295 ) 

296 install_id = _install_id() 

297 if tools and install_id and _core is not None and not self._deliver(_core, tools, counts, install_id): 

298 return 

299 self._last_id = newest_id 

300 self._last_surfaced_id = newest_surfaced_id 

301 self._last_relation_id = newest_relation_id 

302 nudge_count.subtract(counts.nudges_by_kind) 

303 _errors.subtract({tool: counts["errors"] for tool, counts in tools.items()})

305 def report_when_due(self, idle_seconds: float) -> None: 

306 """Send the calls made before a new use began, then once per idle stretch, when it reaches IDLE_SECONDS.""" 

307 if self._new_use_cutoff is not None: 

308 cutoff, self._new_use_cutoff = self._new_use_cutoff, None 

309 self.report(up_to=cutoff) 

310 if idle_seconds < IDLE_SECONDS: 

311 self._reported_this_idle = False 

312 elif not self._reported_this_idle: 

313 self._reported_this_idle = True 

314 self.report()

317class _IdleReporter: 

318 """Lets a report be attempted once per idle stretch and keeps its state across restarts.""" 

319 

320 _state_file: str 

321 

322 def __init__(self) -> None: 

323 """Start with no attempt made in the current idle stretch.""" 

324 self._attempted_this_idle = False 

325 

326 def _state_path(self) -> Path: 

327 return get_data_dir() / self._state_file 

328 

329 def _read_state(self) -> dict[str, Any]: 

330 try: 

331 state = json.loads(self._state_path().read_text(encoding="utf-8")) 

332 except (OSError, ValueError): 

333 return {} 

334 return state if isinstance(state, dict) else {} 

335 

336 def _begin_attempt(self, idle_seconds: float) -> tuple[ModuleType, str] | None: 

337 """Return the core and install id once per idle stretch, and only when telemetry is on.""" 

338 if idle_seconds < IDLE_SECONDS: 

339 self._attempted_this_idle = False 

340 return None 

341 if self._attempted_this_idle: 

342 return None 

343 self._attempted_this_idle = True 

344 install_id = _install_id() 

345 return (_core, install_id) if _core is not None and install_id is not None else None

374 async def report_when_due(self, idle_seconds: float) -> None: 

375 """Score and send once per idle stretch with new searches since the last eval, off the event loop. 

376 

377 Nothing is scored when telemetry is off. A failed send is retried on the next idle stretch. 

378 """ 

379 attempt = self._begin_attempt(idle_seconds) 

380 if attempt is None: 

381 return 

382 core, install_id = attempt 

383 newest = self._newest_search_id() 

384 if newest <= self._read_state().get("search_id", 0): 

385 return 

386 report = await evaluate_cached_async(self._db, k=EVAL_K, min_content_tokens=EVAL_MIN_CONTENT_TOKENS) 

387 delivered = report.query_count > 0 and await asyncio.to_thread(self._deliver, core, report, install_id) 

388 if delivered or not report.query_count: 

389 write_json_atomic(self._state_path(), {"search_id": newest})
