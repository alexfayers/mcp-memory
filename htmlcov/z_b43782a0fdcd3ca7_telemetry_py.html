26 hits: list[tuple[str, str, int]],

28 """Record the entities a ranked search surfaced, for implicit-usefulness telemetry.

43 def register_use(self, project: str, name: str, *, window_seconds: float, max_per_day: int) -> int | None: 

44 """Cast a deterministic implicit-usefulness vote if this entity was recently surfaced. 

45 

46 Consumes the newest not-yet-matched surfacing of ``name`` within ``window_seconds``: 

47 an in-window edit following a search is an observed "this was useful". Casts a bounded 

48 ``+1`` (via the vote_score column directly, not entities.vote, so instrumentation does 

49 not recurse) unless the per-entity daily cap is already reached, in which case the 

50 surfacing is still consumed but no vote fires. Returns the new vote_score, or ``None`` 

51 when there was no eligible surfacing, the cap was hit, or the entity is gone. 

52 """ 

53 with self._conn.transaction(): 

54 surfaced = self._conn.query_one( 

55 "SELECT id FROM surfaced_entities " 

56 "WHERE project = ? AND entity_name = ? AND used_at IS NULL " 

57 "AND surfaced_at >= datetime('now', ?) " 

58 "ORDER BY surfaced_at DESC, id DESC LIMIT 1", 

59 (project, name, seconds_ago(window_seconds)), 

60 ) 

61 if surfaced is None: 

62 return None 

63 

64 self._conn.write( 

65 "UPDATE surfaced_entities SET used_at = CURRENT_TIMESTAMP WHERE id = ?", 

66 (surfaced["id"],), 

67 ) 

68 

69 project_id = get_project_id(self._conn, project) 

70 if project_id is None: 

71 return None 

72 entity_id = get_entity_id(self._conn, name, project_id) 

73 if entity_id is None: 

74 return None 

75 

76 votes_today_row = self._conn.query_one( 

77 "SELECT COUNT(*) AS n FROM surfaced_entities " 

78 "WHERE project = ? AND entity_name = ? AND vote_cast = 1 " 

79 "AND used_at >= date('now')", 

80 (project, name), 

81 ) 

82 assert votes_today_row is not None 

83 if votes_today_row["n"] >= max_per_day: 

84 return None 

85 

86 self._conn.write("UPDATE surfaced_entities SET vote_cast = 1 WHERE id = ?", (surfaced["id"],)) 

87 self._conn.write("UPDATE entities SET vote_score = vote_score + 1 WHERE id = ?", (entity_id,)) 

88 row = self._conn.query_one("SELECT vote_score FROM entities WHERE id = ?", (entity_id,)) 

89 assert row is not None 

90 return int(row["vote_score"])

92 def prune_surfaced(self, retention_days: int) -> int: 

93 """Delete retrieval telemetry older than the retention window, returning the row count. 

94 

95 A negative window means unlimited retention: nothing is deleted and 0 is returned. 

96 """ 

97 if retention_days < 0: 

98 return 0 

99 with self._conn.transaction(): 

100 cursor = self._conn.write( 

101 "DELETE FROM surfaced_entities WHERE surfaced_at < datetime('now', ?)", 

102 (days_ago(retention_days),), 

103 ) 

104 return cursor.rowcount
