1"""The committed measured baseline for MCP read-tool output size, in bytes. 

2 

3Owns the size section's shape, its (de)serialisation and the probe table, within the shared 

4`tests/eval/baseline.json` artefact - see `eval_baseline` for the ranking section and 

5`regen_baseline` for the single entry point that regenerates both. Regenerating it is a 

6separate, explicit act (`just baseline --rebaseline`) - never something a test run does, 

7since a reference value recomputed from the code under test can never fail. 

8""" 

9

27TOOLS = ("search_nodes", "read_graph", "get_entity_with_relations") 

28 

29# Not a real tokenizer - no tiktoken dependency, and it would be the wrong tokenizer family 

30# for a Claude-facing tool anyway. A rough estimate purely to make byte deltas legible as an 

31# approximate token count. 

32_BYTES_PER_TOKEN = 4 

33 

34_REBASELINE_HINT = "run `just baseline --rebaseline` to regenerate it"

37def _search_nodes_probes(fixture: EvalFixture) -> list[int]: 

38 """Return one search_nodes payload size per `_TOPICS` entry, pinned to `fixture.now`.""" 

39 return [ 

40 payload_size(server._prepare_read_result(fixture.db.reads.search(project, term, now=fixture.now))) 

41 for _, project, term in _TOPICS 

42 ]

45def _read_graph_probes(fixture: EvalFixture) -> list[int]: 

46 """Return one read_graph payload size per `_PROJECTS` entry.""" 

47 return [payload_size(server._prepare_read_result(fixture.db.reads.recent(project))) for project, _, _ in _PROJECTS]

50def _get_entity_with_relations_probes(fixture: EvalFixture) -> list[int]: 

51 """Return one get_entity_with_relations payload size per topic's durable-hit entity.""" 

52 return [ 

53 payload_size( 

54 server._prepare_read_result( 

55 fixture.db.reads.get_entity_with_relations(project, fixture.name_for(topic_index, "durable-hit")) 

56 ) 

57 ) 

58 for topic_index, project, _ in _TOPICS 

59 ]

62# tool name -> a probe function returning one payload byte-size per item it measures. Each 

63# probe resolves its own project/query/entity-name arguments from `fixture` at call time. 

64_PROBES: dict[str, Callable[[EvalFixture], list[int]]] = { 

65 "search_nodes": _search_nodes_probes, 

66 "read_graph": _read_graph_probes, 

67 "get_entity_with_relations": _get_entity_with_relations_probes,

71@dataclass(frozen=True) 

72class SizeBaseline: 

73 """A measured output-size baseline: total bytes per tool, at a given fixture entity count.""" 

74 

75 entity_count: int 

76 probe_counts: dict[str, int] 

77 total_bytes: dict[str, int] 

78 

79 @classmethod 

80 def from_probes(cls, entity_count: int, sizes_by_tool: dict[str, list[int]]) -> SizeBaseline: 

81 """Build a canonical baseline from raw per-probe byte sizes, keyed by tool name.""" 

82 return cls( 

83 entity_count=entity_count, 

84 probe_counts={tool: len(sizes) for tool, sizes in sizes_by_tool.items()}, 

85 total_bytes={tool: sum(sizes) for tool, sizes in sizes_by_tool.items()}, 

86 ) 

87 

88 

89def measure(fixture: EvalFixture) -> SizeBaseline: 

90 """Run every probe in `_PROBES` against `fixture` and return the resulting baseline.""" 

91 return SizeBaseline.from_probes( 

92 entity_count=len(fixture.entity_names), 

93 sizes_by_tool={tool: probe(fixture) for tool, probe in _PROBES.items()}, 

94 )

97def load_baseline(path: Path = BASELINE_PATH) -> SizeBaseline: 

98 """Read and validate the size section of the committed baseline artefact at `path`.""" 

99 if not path.exists(): 

100 raise TypeError(f"no size baseline at {path} - {_REBASELINE_HINT}") 

101 try: 

102 payload = json.loads(path.read_text(encoding="utf-8")) 

103 except json.JSONDecodeError as exc: 

104 raise TypeError(f"size baseline at {path} is not valid JSON - {_REBASELINE_HINT}") from exc 

105 if "size" not in payload: 

106 raise TypeError(f"size baseline at {path} has no 'size' section - {_REBASELINE_HINT}") 

107 section_payload = payload["size"] 

108 if not isinstance(section_payload, dict): 

109 raise TypeError(f"size baseline at {path} has a non-dict 'size' section - {_REBASELINE_HINT}") 

110 entity_count = section_payload["entity_count"] 

111 tools = section_payload["tools"] 

112 if set(tools) != set(TOOLS): 

113 raise TypeError(f"size baseline at {path} does not cover exactly {TOOLS} - {_REBASELINE_HINT}") 

114 if entity_count <= 0: 

115 raise TypeError(f"size baseline at {path} was measured at entity_count={entity_count} - {_REBASELINE_HINT}") 

116 probe_counts = {tool: tools[tool]["probes"] for tool in TOOLS} 

117 total_bytes = {tool: tools[tool]["total_bytes"] for tool in TOOLS} 

118 if any(count <= 0 for count in probe_counts.values()): 

119 raise TypeError(f"size baseline at {path} has a non-positive probe count - {_REBASELINE_HINT}") 

120 if any(size <= 0 for size in total_bytes.values()): 

121 raise TypeError(f"size baseline at {path} has a non-positive total_bytes - {_REBASELINE_HINT}") 

122 return SizeBaseline(entity_count=entity_count, probe_counts=probe_counts, total_bytes=total_bytes) 

123 

124 

125def section(baseline: SizeBaseline) -> dict[str, object]: 

126 """Return the size section's payload: fixed key order.""" 

127 return { 

128 "entity_count": baseline.entity_count, 

129 "tools": { 

130 tool: {"probes": baseline.probe_counts[tool], "total_bytes": baseline.total_bytes[tool]} for tool in TOOLS 

131 }, 

132 } 

133 

134 

135def moved(committed: SizeBaseline, measured: SizeBaseline) -> dict[str, tuple[int, int]]: 

136 """Return {tool: (old, new)} for every tool whose total_bytes differs.""" 

137 return { 

138 tool: (committed.total_bytes[tool], measured.total_bytes[tool]) 

139 for tool in TOOLS 

140 if committed.total_bytes[tool] != measured.total_bytes[tool] 

141 } 

142 

143 

144def format_moves(committed: SizeBaseline, measured: SizeBaseline) -> str: 

145 """Render the tool-name list and byte/token delta for every tool whose size moved.""" 

146 drift = moved(committed, measured) 

147 delta = sum(new - old for old, new in drift.values()) 

148 tokens = delta // _BYTES_PER_TOKEN 

149 sign = "+" if delta >= 0 else "" 

150 return ( 

151 f"output size moved for {sorted(drift)}\n" 

152 f"({sign}{delta} bytes, ~{sign}{tokens} tokens estimated at {_BYTES_PER_TOKEN} bytes/token)" 

153 )
