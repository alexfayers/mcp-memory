25 def _validate_entity_data(self, project: str, entity_data: dict[str, object]) -> None: 

26 """Raise ValueError if entity_data does not meet create's input contract.""" 

27 name = entity_data.get("name") 

28 entity_type = entity_data.get("entityType") 

29 observations = entity_data.get("observations") 

30 status = entity_data.get("status") 

31 

32 if not name or not isinstance(name, str): 

33 raise ValueError(f"Entity name must be a non-empty string, got: {name!r}") 

34 if not entity_type or not isinstance(entity_type, str): 

35 raise ValueError(f"Entity type must be a non-empty string, got: {entity_type!r}") 

36 if not isinstance(observations, list) or not observations: 

37 raise ValueError(f"Observations must be a non-empty list for entity '{name}'") 

38 for obs in observations: 

39 if not isinstance(obs, str) or not obs: 

40 raise ValueError(f"Each observation must be a non-empty string for entity '{name}', got: {obs!r}") 

41 

42 if status is not None and status not in VALID_STATUSES: 

43 raise ValueError(f"Invalid status '{status}' for entity '{name}'. Must be one of: {VALID_STATUSES}") 

44 

45 if get_strict_policy_enabled() and project != "global" and entity_type == "user-preferences": 

46 raise ValueError( 

47 "Project-scoped 'user-preferences' entities are forbidden when " 

48 "MCP_MEMORY_STRICT_POLICY is enabled; store them in global scope instead." 

49 )

51 def create(self, project: str, entities: list[dict[str, object]]) -> None: 

52 """Create new entities with observations. Raises if any name already exists.""" 

53 project_id = get_or_create_project_id(self._conn, project) 

54 

55 with self._conn.transaction(): 

56 for entity_data in entities: 

57 self._validate_entity_data(project, entity_data) 

58 name = entity_data.get("name") 

59 entity_type = entity_data.get("entityType") 

60 observations = cast("list[str]", entity_data.get("observations")) 

61 observations = [strip_today_date_prefix(obs) for obs in observations] 

62 status = entity_data.get("status") 

63 

64 if get_entity_id(self._conn, str(name), project_id) is not None: 

65 raise ValueError( 

66 f"Entity '{name}' already exists in project '{project}'. " 

67 "create_entities only creates new entities - use add_observations to " 

68 "append, or another update method to modify it." 

69 ) 

70 

71 entity_type_id = get_or_create_entity_type_id(self._conn, str(entity_type)) 

72 self.purge_tombstone(str(name), project, project_id) 

73 

74 cursor = self._conn.write( 

75 "INSERT INTO entities (name, entity_type_id, project_id, status) VALUES (?, ?, ?, ?)", 

76 (name, entity_type_id, project_id, status), 

77 ) 

78 entity_id = cast("int", cursor.lastrowid) 

79 

80 self._conn.write_many( 

81 "INSERT INTO observations (entity_id, content, content_hash) VALUES (?, ?, ?)", 

82 [(entity_id, obs, hash_observation(obs)) for obs in observations], 

83 )

85 def rename(self, project: str, old_name: str, new_name: str) -> None: 

86 """Rename an entity in place, preserving its relations and observations.""" 

87 project_id = get_or_create_project_id(self._conn, project) 

88 entity_id = get_entity_id(self._conn, old_name, project_id) 

89 if entity_id is None: 

90 raise ValueError(f"Entity '{old_name}' not found in project '{project}'") 

91 

92 if get_entity_id(self._conn, new_name, project_id) is not None: 

93 raise ValueError(f"Entity '{new_name}' already exists in project '{project}'") 

94 

95 if project == "global": 

96 conflict = self.exists_outside(new_name, "global") 

97 if conflict: 

98 raise ValueError(f"Entity '{new_name}' already exists in project '{conflict}'") 

99 elif self.exists_in(new_name, "global"): 

100 raise ValueError(f"Entity '{new_name}' already exists in global scope") 

101 

102 with self._conn.transaction(): 

103 refresh_for_entity(self._conn, entity_id, delete=True) 

104 self._conn.write("UPDATE entities SET name = ? WHERE id = ?", (new_name, entity_id)) 

105 refresh_for_entity(self._conn, entity_id, delete=False)

107 def move_cross_scope(self, source_project: str, target_project: str, name: str) -> list[Relation]: 

108 """Move an entity to another scope, dropping and returning its cross-scope relations.""" 

109 source_id = get_or_create_project_id(self._conn, source_project) 

110 target_id = get_or_create_project_id(self._conn, target_project) 

111 entity_id = get_entity_id(self._conn, name, source_id) 

112 if entity_id is None: 

113 raise ValueError(f"Entity '{name}' not found in project '{source_project}'") 

114 

115 if get_entity_id(self._conn, name, target_id) is not None: 

116 raise ValueError(f"Entity '{name}' already exists in project '{target_project}'") 

117 

118 with self._conn.transaction(): 

119 rows = self._conn.query_all( 

120 "SELECT e_src.name AS source, e_tgt.name AS target, rt.name AS relation_type " 

121 "FROM relations r " 

122 "JOIN entities e_src ON r.source_id = e_src.id " 

123 "JOIN entities e_tgt ON r.target_id = e_tgt.id " 

124 "JOIN relation_types rt ON r.relation_type_id = rt.id " 

125 "WHERE r.source_id = ? OR r.target_id = ?", 

126 (entity_id, entity_id), 

127 ) 

128 dropped = [build_relation(row) for row in rows] 

129 self._conn.write( 

130 "DELETE FROM relations WHERE source_id = ? OR target_id = ?", 

131 (entity_id, entity_id), 

132 ) 

133 refresh_for_entity(self._conn, entity_id, delete=True) 

134 self._conn.write("UPDATE entities SET project_id = ? WHERE id = ?", (target_id, entity_id)) 

135 refresh_for_entity(self._conn, entity_id, delete=False) 

136 

137 return dropped

139 def exists_in(self, name: str, project: str) -> bool: 

140 """Check if an entity name exists in a specific project scope.""" 

141 row = self._conn.query_one( 

142 "SELECT 1 FROM entities e JOIN projects p ON e.project_id = p.id " 

143 "WHERE e.name = ? AND p.name = ? AND e.deleted_at IS NULL", 

144 (name, project), 

145 ) 

146 return row is not None

148 def exists_outside(self, name: str, project: str) -> str | None: 

149 """Return the first project where this entity exists, excluding the given project.""" 

150 row = self._conn.query_one( 

151 "SELECT p.name FROM entities e JOIN projects p ON e.project_id = p.id " 

152 "WHERE e.name = ? AND p.name != ? AND e.deleted_at IS NULL", 

153 (name, project),

155 return row["name"] if row else None

157 def set_status(self, project: str, name: str, status: EntityStatus | None) -> int: 

158 """Set or clear the status of an entity and return its observation count.""" 

159 if status is not None and status not in VALID_STATUSES: 

160 raise ValueError(f"Invalid status '{status}'. Must be one of: {VALID_STATUSES}") 

161 

162 project_id = get_or_create_project_id(self._conn, project) 

163 entity_id = get_entity_id(self._conn, name, project_id) 

164 if entity_id is None: 

165 raise ValueError(f"Entity '{name}' not found in project '{project}'") 

166 

167 with self._conn.transaction(): 

168 self._conn.write("UPDATE entities SET status = ? WHERE id = ?", (status, entity_id)) 

169 row = self._conn.query_one("SELECT COUNT(*) AS n FROM observations WHERE entity_id = ?", (entity_id,)) 

170 assert row is not None 

171 return int(row["n"])

173 def vote(self, project: str, name: str, vote: int) -> int: 

174 """Apply a usefulness vote to an entity and return its new net score. 

175 

176 Any nonzero integer up to MAX_VOTE_MAGNITUDE in magnitude sets the vote's strength. 

177 Deliberately updates only vote_score, leaving updated_at untouched, so a vote (a 

178 relevance signal) is not mistaken for a content change by recency ranking. 

179 """ 

180 validate_vote(vote) 

181 

182 project_id = get_or_create_project_id(self._conn, project) 

183 entity_id = get_entity_id(self._conn, name, project_id) 

184 if entity_id is None: 

185 raise ValueError(f"Entity '{name}' not found in project '{project}'") 

186 

187 with self._conn.transaction(): 

188 self._conn.write( 

189 "UPDATE entities SET vote_score = vote_score + ? WHERE id = ?", 

190 (vote, entity_id), 

191 ) 

192 row = self._conn.query_one("SELECT vote_score FROM entities WHERE id = ?", (entity_id,)) 

193 assert row is not None 

194 return int(row["vote_score"])

196 def delete(self, project: str, name: str) -> None: 

197 """Delete an entity, cascading outgoing relations but blocking on incoming.""" 

198 project_id = get_or_create_project_id(self._conn, project) 

199 entity_id = get_entity_id(self._conn, name, project_id) 

200 if entity_id is None: 

201 raise ValueError(f"Entity '{name}' not found in project '{project}'") 

202 

203 incoming = self._conn.query_all( 

204 "SELECT e.name FROM relations r " 

205 "JOIN entities e ON r.source_id = e.id " 

206 "WHERE r.target_id = ? AND r.source_id != ?", 

207 (entity_id, entity_id), 

208 ) 

209 if incoming: 

210 sources = [row["name"] for row in incoming] 

211 raise ValueError(f"Cannot delete '{name}': {len(sources)} incoming relation(s) from: " + ", ".join(sources)) 

212 

213 orphaned = orphaned_neighbors_if_entity_deleted(self._conn, entity_id) 

214 if orphaned: 

215 raise ValueError( 

216 f"Cannot delete '{name}': it would orphan non-structural " 

217 f"entit{'y' if len(orphaned) == 1 else 'ies'}: {', '.join(orphaned)}" 

218 ) 

219 

220 with self._conn.transaction(): 

221 self.hard_delete_row(entity_id, project, name)

223 def hard_delete_row(self, entity_id: int, project: str, name: str) -> None: 

224 """Hard-delete an entity and its observations, outgoing relations and telemetry. 

225 

226 Executes within the caller's transaction (no commit of its own), so it is safe 

227 to call from inside another ``with self._conn.transaction():`` block. Deletes 

228 relations in both directions so a force-delete (e.g. the purge of a merged-away 

229 entity that still holds an incoming edge) cannot orphan a relation row against 

230 the foreign key. 

231 """ 

232 self._conn.write("DELETE FROM observations WHERE entity_id = ?", (entity_id,)) 

233 self._conn.write("DELETE FROM relations WHERE source_id = ? OR target_id = ?", (entity_id, entity_id)) 

234 self._conn.write( 

235 "DELETE FROM surfaced_entities WHERE project = ? AND entity_name = ?", 

236 (project, name), 

237 ) 

238 self._conn.write("DELETE FROM entities WHERE id = ?", (entity_id,))

240 def purge_tombstone(self, name: str, project: str, project_id: int) -> None: 

241 """Hard-remove a soft-deleted row holding the (name, project) slot before a re-create. 

242 

243 The UNIQUE(name, project_id) constraint means a tombstone would otherwise block 

244 inserting a fresh entity with the same name. Runs within the caller's transaction. 

245 """ 

246 tombstone_id = get_entity_id(self._conn, name, project_id, include_deleted=True) 

247 if tombstone_id is not None: 

248 self.hard_delete_row(tombstone_id, project, name)

250 def restore(self, project: str, name: str) -> None: 

251 """Clear an entity's soft-deleted mark, making it visible to reads again.""" 

252 project_id = get_or_create_project_id(self._conn, project) 

253 entity_id = get_entity_id(self._conn, name, project_id, include_deleted=True) 

254 if entity_id is None: 

255 raise ValueError(f"Entity '{name}' not found in project '{project}'") 

256 with self._conn.transaction(): 

257 self._conn.write("UPDATE entities SET deleted_at = NULL WHERE id = ?", (entity_id,))

259 def merge(self, project: str, source: str, target: str) -> dict[str, int]: 

260 """Fold a source entity into a target, then soft-delete the source. 

261 

262 Additive on the target: the source's observations are copied over (deduped by 

263 exact content, keeping their votes and timestamps), and equivalent relations are 

264 recreated on the target (skipping any that would become a target->target 

265 self-loop). The target keeps the higher of the two vote scores. The source is 

266 soft-deleted rather than removed, so the merge is reversible via restore. 

267 """ 

268 if source == target: 

269 raise ValueError("Cannot merge an entity into itself") 

270 project_id = get_or_create_project_id(self._conn, project) 

271 source_id = get_entity_id(self._conn, source, project_id) 

272 if source_id is None: 

273 raise ValueError(f"Source entity '{source}' not found in project '{project}'") 

274 target_id = get_entity_id(self._conn, target, project_id) 

275 if target_id is None: 

276 raise ValueError(f"Target entity '{target}' not found in project '{project}'") 

277 

278 with self._conn.transaction(): 

279 obs = self._conn.write( 

280 "INSERT INTO observations " 

281 "(entity_id, content, vote_score, created_at, content_hash) " 

282 "SELECT ?, content, vote_score, created_at, content_hash FROM observations " 

283 "WHERE entity_id = ? AND content NOT IN " 

284 "(SELECT content FROM observations WHERE entity_id = ?)", 

285 (target_id, source_id, target_id), 

286 ) 

287 outgoing = self._conn.write( 

288 "INSERT OR IGNORE INTO relations (source_id, target_id, relation_type_id) " 

289 "SELECT ?, target_id, relation_type_id FROM relations " 

290 "WHERE source_id = ? AND target_id != ?", 

291 (target_id, source_id, target_id), 

292 ) 

293 incoming = self._conn.write( 

294 "INSERT OR IGNORE INTO relations (source_id, target_id, relation_type_id) " 

295 "SELECT source_id, ?, relation_type_id FROM relations " 

296 "WHERE target_id = ? AND source_id != ?", 

297 (target_id, source_id, target_id), 

298 ) 

299 self._conn.write( 

300 "UPDATE entities SET vote_score = " 

301 "MAX(vote_score, (SELECT vote_score FROM entities WHERE id = ?)) WHERE id = ?", 

302 (source_id, target_id), 

303 ) 

304 self._conn.write("UPDATE entities SET deleted_at = CURRENT_TIMESTAMP WHERE id = ?", (source_id,)) 

305 return { 

306 "observations_merged": obs.rowcount, 

307 "relations_repointed": outgoing.rowcount + incoming.rowcount, 

308 }
