28def _baseline(**overrides: int) -> SizeBaseline: 

29 """Build a valid `SizeBaseline`, overridable per tool's total_bytes.""" 

30 probe_counts = {"search_nodes": 6, "read_graph": 12, "get_entity_with_relations": 6} 

31 total_bytes = {"search_nodes": 4000, "read_graph": 9000, "get_entity_with_relations": 2500} 

32 total_bytes.update(overrides) 

33 return SizeBaseline(entity_count=130, probe_counts=probe_counts, total_bytes=total_bytes)

125 ) -> None:
