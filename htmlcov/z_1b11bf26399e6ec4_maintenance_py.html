26# Vote score at or below which an orphan becomes eligible for autonomous GC. agent.py imports 

27# this so the dream's saturation-floor prose and the GC reap threshold share one value. 

28GC_DOWNVOTE_FLOOR = -10

31class Maintenance: 

32 """Best-effort sweeps: telemetry pruning, orphan GC, tombstone purge, and archival.""" 

33 

34 def __init__(self, connection: Connection, entities: EntityRepository, telemetry: TelemetryRepository) -> None: 

35 self._conn = connection 

36 self._entities = entities 

37 self._telemetry = telemetry 

38 

39 def backfill_observation_hashes(self) -> None: 

40 """Populate content_hash for any observation missing one. No-op after the first run.""" 

41 rows = self._conn.query_all("SELECT id, content FROM observations WHERE content_hash IS NULL") 

42 if not rows: 

43 return 

44 with self._conn.transaction(): 

45 for row in rows: 

46 self._conn.write( 

47 "UPDATE observations SET content_hash = ? WHERE id = ?", 

48 (hash_observation(row["content"]), row["id"]), 

49 ) 

50 

51 def _purge_soft_deleted(self, grace_days: int) -> int: 

52 """Hard-delete entities soft-deleted longer than the grace window, returning the count.""" 

53 rows = self._conn.query_all( 

54 "SELECT e.id, e.name, p.name AS project FROM entities e " 

55 "JOIN projects p ON e.project_id = p.id " 

56 "WHERE e.deleted_at IS NOT NULL AND e.deleted_at < datetime('now', ?)", 

57 (days_ago(grace_days),), 

58 ) 

59 with self._conn.transaction(): 

60 for row in rows: 

61 self._entities.hard_delete_row(row["id"], row["project"], row["name"]) 

62 return len(rows) 

63 

64 def _gc_downvoted_orphans(self, threshold: int = GC_DOWNVOTE_FLOOR) -> int: 

65 """Soft-delete downvoted orphan entities, returning the number reaped. 

66 

67 Reaps an entity that is all of: live, at or below the downvote floor, not a 

68 relation-exempt type, and with no live incoming relation. Removal is a reversible 

69 soft-delete, so the grace-window purge stays the sole path to permanent removal. 

70 """ 

71 with self._conn.transaction(): 

72 cursor = self._conn.write( 

73 f"UPDATE entities SET deleted_at = CURRENT_TIMESTAMP " 

74 f"WHERE deleted_at IS NULL " 

75 f"AND vote_score <= ? " 

76 f"AND entity_type_id NOT IN " 

77 f"(SELECT id FROM entity_types WHERE name IN " 

78 f"({placeholders(len(STRUCTURAL_ENTITY_TYPES))})) " 

79 f"AND NOT EXISTS (" 

80 f"SELECT 1 FROM relations r " 

81 f"JOIN entities src ON r.source_id = src.id " 

82 f"WHERE r.target_id = entities.id AND src.deleted_at IS NULL)", 

83 (threshold, *STRUCTURAL_ENTITY_TYPES), 

84 ) 

85 return cursor.rowcount

87 def _archive_stale_entities(self, threshold_days: int = ARCHIVE_STALE_DAYS) -> int: 

88 """Auto-archive stale resolved entities, returning the number archived. 

89 

90 Archives an entity that is all of: live, status='resolved', last updated more than 

91 threshold_days ago, and outside the never-evict set - entities ever acted on after 

92 being surfaced (a row in surfaced_entities with used_at set). Any write or acted-on 

93 retrieval bumps updated_at, so renewal is already handled elsewhere. 

94 """ 

95 with self._conn.transaction(): 

96 cursor = self._conn.write( 

97 "UPDATE entities SET status = 'archived' " 

98 "WHERE deleted_at IS NULL " 

99 "AND status = 'resolved' " 

100 "AND updated_at < datetime('now', ?) " 

101 "AND name NOT IN " 

102 "(SELECT entity_name FROM surfaced_entities WHERE used_at IS NOT NULL)", 

103 (days_ago(threshold_days),), 

104 ) 

105 return cursor.rowcount

107 def run_sweeps(self) -> None: 

108 """Run best-effort maintenance in a fixed order, tolerating lock contention. 

109 

110 Order: prune tool-call telemetry, prune surfaced-entity telemetry, then - each gated 

111 by its own config flag - orphan GC, soft-delete purge, and stale-entity archival. A 

112 short-lived process (e.g. the CLI) opening its own connection can race a long-running 

113 service's write and hit "database is locked" past busy_timeout. None of this is 

114 required for the caller's own request to succeed, so the whole block is skipped 

115 silently on contention rather than crashing the command. Does NOT run the 

116 observation-hash backfill: the composition root calls that separately and 

117 unprotected, since it is required for correctness rather than best-effort. 

118 """ 

119 try: 

120 self._telemetry.prune_tool_calls(get_call_metrics_retention_days()) 

121 self._telemetry.prune_surfaced(get_surfaced_retention_days()) 

122 self._run_gated_sweeps() 

123 except sqlite3.OperationalError: 

124 pass 

125 

126 def _run_gated_sweeps(self) -> None: 

127 """Run the config-gated sweeps: orphan GC, soft-delete purge, and stale archival.""" 

128 if get_gc_enabled(): 

129 self._gc_downvoted_orphans() 

130 if get_purge_enabled(): 

131 self._purge_soft_deleted(get_purge_grace_days()) 

132 if get_archive_enabled(): 

133 self._archive_stale_entities()
