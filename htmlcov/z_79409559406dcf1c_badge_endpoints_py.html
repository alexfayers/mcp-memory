20def build_parser() -> argparse.ArgumentParser:

28def _endpoint(label: str, message: str) -> str: 

29 """Return the shields.io endpoint JSON for a blue badge.""" 

30 return json.dumps({"schemaVersion": 1, "label": label, "message": message, "color": "blue"}) + "\n" 

31 

32 

33def main(argv: Sequence[str] | None = None) -> int: 

34 """Write `size.json` and `ranking.json` for the artefact into the output directory.""" 

35 args = build_parser().parse_args(argv) 

36 artefact = load_artefact(args.artefact) 

37 args.out_dir.mkdir(parents=True, exist_ok=True) 

38 total_kb = sum(artefact.sizes.values()) / _BYTES_PER_KB 

39 (args.out_dir / "size.json").write_text(_endpoint("tool output", f"{total_kb:.1f} kB"), encoding="utf-8") 

40 recall = artefact.ranking[_RECALL_METRIC] 

41 (args.out_dir / "ranking.json").write_text( 

42 _endpoint(f"recall@{artefact.k}", f"{recall:.{_RANKING_DECIMALS}f}"), encoding="utf-8" 

43 )
