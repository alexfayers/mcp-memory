1"""The measured baseline for MCP read-tool latency, in milliseconds per call. 

2 

3Owns the latency section's shape and serialisation within the shared `tests/eval/baseline.json` 

4artefact, and times the same probes as `size_baseline`, so each figure covers the tool call and 

5the sizing of its payload. See `regen_baseline` for the entry point that measures and writes every 

6section. Timings vary between runs and machines, so `compare_baseline` reports them without gating. 

7""" 

8 

9from __future__ import annotations 

10 

11from dataclasses import dataclass 

12import statistics 

13import time 

14from typing import TYPE_CHECKING 

15 

16from tests.eval.size_baseline import _PROBES, TOOLS 

17 

18if TYPE_CHECKING: 

19 from .eval_fixture import EvalFixture 

20 

21_REPEATS = 5 

22_MS_PER_SECOND = 1000 

23_DP = 3 

24 

25 

26@dataclass(frozen=True) 

27class LatencyBaseline: 

28 """A measured latency baseline: the median milliseconds per call, per tool.""" 

29 

30 median_ms: dict[str, float] 

31 

32 

33def measure(fixture: EvalFixture, *, repeats: int = _REPEATS) -> LatencyBaseline: 

34 """Time every probe in `_PROBES` against `fixture` and return its median per-call milliseconds.""" 

35 medians: dict[str, float] = {} 

36 for tool, probe in _PROBES.items(): 

37 per_call_ms: list[float] = [] 

38 for _ in range(repeats): 

39 started = time.perf_counter() 

40 calls = len(probe(fixture)) 

41 per_call_ms.append((time.perf_counter() - started) * _MS_PER_SECOND / calls) 

42 medians[tool] = round(statistics.median(per_call_ms), _DP) 

43 return LatencyBaseline(median_ms=medians) 

44 

45 

46def section(baseline: LatencyBaseline) -> dict[str, object]: 

47 """Return the latency section's payload: fixed key order.""" 

48 return {"tools": {tool: {"median_ms": baseline.median_ms[tool]} for tool in TOOLS}}
